function isRealAddress(addr){
  return !addr.includes('Citywide');
}

function sleep(ms){ return new Promise(r=>setTimeout(r,ms)); }

function getDirections(destination){
  return new Promise((resolve)=>{
    const url='https://maps.googleapis.com/maps/api/directions/json'
      +'?origin='+encodeURIComponent(ORIGIN)
      +'&destination='+encodeURIComponent(destination)
      +'&mode=transit'
      +'&key='+API_KEY;

    https.get(url,res=>{
      let data='';
      res.on('data',chunk=>data+=chunk);
      res.on('end',()=>{
        try{
          const json=JSON.parse(data);
          if(json.status!=='OK'){
            console.log('  API status:',json.status);
            resolve('Ask staff for directions to this location.');
            return;
          }
          const steps=json.routes[0].legs[0].steps;
          const text=steps.map(s=>{
            const instr=s.html_instructions
              .replace(/<b>/g,'').replace(/<\/b>/g,'')
              .replace(/<div[^>]*>/g,' — ').replace(/<\/div>/g,'')
              .replace(/<[^>]+>/g,'').replace(/\s+/g,' ').trim();
            if(s.travel_mode==='TRANSIT'&&s.transit_details){
              const td=s.transit_details;
              return instr
                +' — take the '+td.line.short_name
                +' toward '+td.headsign
                +' ('+td.num_stops+' stop'+(td.num_stops>1?'s':'')+')'
                +', exit at '+td.arrival_stop.name;
            }
            return instr;
          }).join('. ');
          resolve(text);
        }catch(e){
          resolve('Ask staff for directions to this location.');
        }
      });
    }).on('error',()=>resolve('Ask staff for directions to this location.'));
  });
}

async function main(){
  const results=[];

  for(let i=0;i<addresses.length;i++){
    const addr=addresses[i];
    let directions;

    if(!isRealAddress(addr)){
      if(addr.includes('Online')) directions='Available online — no travel needed.';
      else directions='Call or text — no travel needed.';
      console.log(`[${i+1}/${addresses.length}] Skipping: ${addr}`);
    } else {
      console.log(`[${i+1}/${addresses.length}] Getting directions for: ${addr}`);
      directions=await getDirections(addr);
      console.log(`  → ${directions.slice(0,100)}...`);
      await sleep(300);
    }

    results.push({
      row: i+1,
      address: addr,
      directions
    });
  }

  fs.writeFileSync('directions_output.json', JSON.stringify(results,null,2));

  // Also write a simple CSV for easy pasting into sheets
  const csv=results.map(r=>
    '"'+r.row+'","'+r.address.replace(/"/g,'""')+'","'+r.directions.replace(/"/g,'""')+'"'
  ).join('\n');
  fs.writeFileSync('directions_output.csv','row,address,directions\n'+csv);

  console.log('\n✅ Done!');
  console.log('directions_output.json — full results');
  console.log('directions_output.csv — paste into Google Sheets');
}

main();
